problem 1: Electricity Bill Calculator

def problem1_electricity_bill():
    # Get total units consumed from the user
    units = int(input("Enter total units consumed: "))

    # Slab-wise billing:
    # First 100 units @ ₹5, next 100 units @ ₹7, anything above 200 @ ₹10
    if units <= 100:
        bill = units * 5
    elif units <= 200:
        bill = (100 * 5) + (units - 100) * 7
    else:
        bill = (100 * 5) + (100 * 7) + (units - 200) * 10

    # If bill crosses ₹2000, add a 5% surcharge on top of it
    if bill > 2000:
        final_amount = bill + (bill * 0.05)
    else:
        final_amount = bill

    # Enforce a minimum bill amount of ₹500
    if final_amount < 500:
        final_amount = 500

    # Display results
    print(f"\nTotal Units       : {units}")
    print(f"Bill Amount       : ₹{bill:.2f}")
    print(f"Final Payable Amt : ₹{final_amount:.2f}")
